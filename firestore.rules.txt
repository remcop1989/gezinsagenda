rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    match /households/{householdId} {

      match /meta/{docId} {
        allow read: if request.auth != null && docId == 'main';
        allow write: if request.auth != null
          && docId == 'main'
          && request.resource.data.keys().hasOnly(['familyMembers', 'settings'])
          && request.resource.data.familyMembers is list
          && request.resource.data.familyMembers.size() < 50
          && (!('settings' in request.resource.data) || request.resource.data.settings is map);
      }

      match /events/{eventId} {
        allow read: if request.auth != null;
        allow create, update: if request.auth != null
          && request.resource.data.size() < 30
          && eventId == request.resource.data.id;
        allow delete: if request.auth != null;
      }

      match /lists/{listId} {
        allow read: if request.auth != null;
        allow create, update: if request.auth != null
          && request.resource.data.size() < 30
          && listId == request.resource.data.id;
        allow delete: if request.auth != null;
      }
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
